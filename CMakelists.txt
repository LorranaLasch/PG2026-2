cmake_minimum_required(VERSION 3.11)
project(PG2026-2)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED True)

# Diretórios de cabeçalhos
include_directories(${CMAKE_SOURCE_DIR}/include)
include_directories(${CMAKE_SOURCE_DIR}/Common)
include_directories(${CMAKE_SOURCE_DIR}/include/glad)
include_directories(${CMAKE_SOURCE_DIR}/include/glm)
include_directories(${CMAKE_SOURCE_DIR}/include/GLFW)
include_directories(${CMAKE_SOURCE_DIR}/src/TrabalhoPraticoGA/include)

# Lista de exercícios e projetos
set(EXERCISES
    exemplos/HelloTriangle
    exemplos/HelloOrtho
    exercicios/lista1/exercicio_1
    exercicios/lista1/exercicio_2
    exercicios/lista1/exercicio_3
    exercicios/lista1/exercicio_4
    exercicios/lista2/exercicio_1_a_5
    exercicios/lista2/exercicio_6
    exercicios/lista3/exercicio_mouse_triangulos
    exercicios/lista3/exercicio_transformacoes
    exercicios/lista3/exercicio_movimento_teclado
    exercicios/lista3/exercicio_screensaver_pong
    TrabalhoPraticoGA
)

add_compile_options(-Wno-pragmas)

# Bibliotecas do sistema
if(WIN32)
    link_directories(${CMAKE_SOURCE_DIR}/Common/lib)
    set(SYS_LIBS opengl32 gdi32 glfw3)
elseif(APPLE)
    find_library(OpenGL_LIBRARY OpenGL)
    set(SYS_LIBS ${OpenGL_LIBRARY} glfw)
else()
    find_package(OpenGL REQUIRED)
    set(SYS_LIBS ${OPENGL_gl_LIBRARY} glfw dl pthread)
endif()

# Arquivo da GLAD
set(GLAD_C_FILE "${CMAKE_SOURCE_DIR}/Common/glad.c")

foreach(EXERCISE ${EXERCISES})
    get_filename_component(EXE_NAME ${EXERCISE} NAME)
    file(GLOB EXE_SOURCES src/${EXERCISE}/*.cpp)

    add_executable(${EXE_NAME} ${EXE_SOURCES} ${GLAD_C_FILE})
    target_link_libraries(${EXE_NAME} ${SYS_LIBS})
endforeach()
